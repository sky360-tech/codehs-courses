public class SocialTags
{
    private String[] allowedTags;

    public SocialTags(String[] allowedTags)
    {
        this.allowedTags = allowedTags;
    }

    // Part (a): Check if a tag matches any allowed tag
    private boolean isAllowedTag(String tag)
    {
        for (String allowed : allowedTags)
        {
            if (allowed.equals(tag))
            {
                return true;
            }
        }
        return false;
    }

    // Part (b): Count how many userTags are not allowed
    private int countDisallowedTags(String[] userTags)
    {
        int count = 0;
        for (String tag : userTags)
        {
            if (!isAllowedTag(tag))
            {
                count++;
            }
        }
        return count;
    }

    // Part (c): Return exactly-sized array of disallowed tags
    public String[] getDisallowedTags(String[] userTags)
    {
        String[] disallowed = new String[countDisallowedTags(userTags)];
        int index = 0;
        for (String tag : userTags)
        {
            if (!isAllowedTag(tag))
            {
                disallowed[index] = tag;
                index++;
            }
        }
        return disallowed;
    }
}
