public class UserAccount
{
    // (a) Instance variable declarations with appropriate access
    private String username;
    private String email;
    private int birthYear;

    // (b) Constructor with parameters
    public UserAccount(String uName, String uEmail, int bYear)
    {
        username = uName;
        email = uEmail;
        birthYear = bYear;
    }

    // (b) No-parameter constructor
    public UserAccount()
    {
        username = "none";
        email = "none";
        birthYear = 2000;
    }

    // Returns the user's username
    public String getUsername()
    {
        return username;
    }
    
    // Returns the user's email
    public String getEmail()
    {
        return email;
    }

    // Computes and returns the user's current age
    public int getAge(int currentYear)
    {
        return currentYear - birthYear;
    }
}
